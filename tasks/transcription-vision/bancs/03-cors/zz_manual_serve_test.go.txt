package gui

import (
	"os"
	"testing"
	"time"
)

func TestZZManualServe(t *testing.T) {
	out := os.Getenv("CORS_URL_FILE")
	if out == "" {
		t.Skip()
	}
	var h mediaHost
	u, err := h.register(os.Getenv("CORS_VIDEO"))
	if err != nil {
		t.Fatal(err)
	}
	_ = os.WriteFile(out, []byte(u), 0o644)
	for {
		if _, err := os.Stat(os.Getenv("CORS_STOP")); err == nil {
			return
		}
		time.Sleep(200 * time.Millisecond)
	}
}
